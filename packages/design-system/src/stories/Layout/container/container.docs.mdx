## Props

```ts
export type ContainerSize =
  | "sm" // 640px
  | "md" // 768px
  | "lg" // 1024px
  | "xl" // 1280px
  | "2xl" // 1536px
  | "full" // 100%
  | number
  | string;

export type ContainerPadding =
  | keyof DefaultTheme["spacing"]
  | number
  | string
  | "none";

export interface ContainerProps {
  as?: ElementType;
  children?: ReactNode;
  size?: ContainerSize; // default: "xl"
  padding?: ContainerPadding; // default: 16
  centered?: boolean; // default: false
  fullWidth?: boolean; // default: false
}
```
